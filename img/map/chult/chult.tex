\newcommand{\MapMarker}[3]{%
  \pgfmathsetmacro{\mx}{\MapX + (#2)*\MapW}%
  \pgfmathsetmacro{\my}{\MapY + (#3)*\MapH}%
  \node[
    circle,
    fill=markerred,
    draw=white,
    line width=0.7pt,
    minimum size=6.8mm,
    inner sep=0pt,
    font=\bfseries\small,
    text=white
  ] at ([xshift=\mx mm,yshift=\my mm]current page.south west) {#1};%
}

\newcommand{\LegendNumber}[1]{%
  \tikz[baseline=(n.base)]\node[
    circle,
    fill=markerred,
    draw=markertext,
    line width=0.45pt,
    minimum size=6.5mm,
    inner sep=0pt,
    font=\bfseries\small,
    text=white
  ] (n) {#1};%
}
